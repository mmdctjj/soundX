# AudioDock HarmonyOS Client — P2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the **content → search → like** surface for the HarmonyOS client: the five detail pages, a text-only Search page, and a `LikesHistoryService` that writes local-first and syncs in the background, all on top of the P0/P1 build pipeline, storage, network, i18n, theme, auth, and AVPlayer.

**Architecture:** Native ArkTS (strict). All five detail pages use `router.pushUrl` + `params.id` and an inline `@Builder` shell that calls the matching `xxxApi.getById(id)`; on tap, they dispatch `playerStore.playTrackList` so the existing P1 player + lock-screen + notification path is unchanged. `SearchPage` reuses the existing `trackApi.search` / `albumApi.search` / `artistApi.search` / `playlistApi.search` with a 300ms debounce and four `Tabs`. `LikesHistoryService` is a singleton service (not a store) that wraps the existing `rdbStore` (write local-first) and `httpClient` (drain a sync queue on a 5s timer). No new HAR modules; no new npm packages; no pattern drift from P0/P1.

**Tech Stack:**
- HarmonyOS NEXT (API 12+), compileSdk 6.0.2(22), target 5.1.1(19), runtimeOS HarmonyOS

- `@ohos.net.http`, `@ohos.data.preferences`, `@ohos.data.relationalStore`

```typescript
import router from '@ohos.router';
import { buildTheme, CommonNavBar, CommonButton, EmptyView, SkeletonBlock } from 'features_ui';
import { t } from 'features_i18n';
import { formatDuration } from 'audiodock_common';
import { albumApi } from 'features_network';
import type { AlbumDto, TrackDto } from 'features_network';
import { resolveTrackToPlayer } from '../services/trackResolver';
import { playerStore } from '../context/PlayerStore';
import type { Track } from '../context/PlayerStore';

interface AlbumDetail extends AlbumDto { tracks: TrackDto[] }

@Entry
@Component
struct AlbumDetailPage {
  @State data: AlbumDetail | null = null;
  @State loading: boolean = true;
  @State error: string = '';
  @State liked: boolean = false;
  private id: string = '';
  private theme = buildTheme('light');

  aboutToAppear(): void {
    const p = router.getParams() as Record<string, string>;
    this.id = p['id'] ?? '';
    this.load();
  }

  build() {
    Column() {
      CommonNavBar({ title: this.data?.name ?? '', theme: this.theme,
        onBack: () => router.back() });
      if (this.loading) {
        Column({ space: 12 }) {
          SkeletonBlock({ blockHeight: 200, theme: this.theme });
          SkeletonBlock({ blockWidth: '60%', blockHeight: 24, theme: this.theme });
          SkeletonBlock({ blockWidth: '40%', blockHeight: 16, theme: this.theme });
        }.padding(16);
      } else if (this.error || !this.data) {
        Column() {
          EmptyView({ message: this.error || t('detail.retry_failed'), theme: this.theme });
          CommonButton({ label: t('detail.retry'), theme: this.theme, onButtonClick: () => this.load() });
        }.padding(16);
      } else {
        this.content();
      }
    }.width('100%').height('100%').backgroundColor(this.theme.colors.background);
  }

  @Builder content() {
    Column({ space: 12 }) {
      Text(this.data!.name).fontSize(22).fontWeight(FontWeight.Bold);
      Text(`${this.data!.artist ?? ''} · ${this.data!.releaseDate ?? ''}`)
        .fontSize(13).fontColor(this.theme.colors.textSecondary);
      if (this.data!.description) {
        Text(this.data!.description)
          .fontSize(13).fontColor(this.theme.colors.textSecondary)
          .maxLines(4);
      }
      CommonButton({
        label: this.liked ? t('detail.unlike') : t('detail.like'),
        theme: this.theme, variant: this.liked ? 'secondary' : 'primary',
        onButtonClick: () => { /* wired in T19 fix-up commit */ },
      });
      List() {
        ForEach(this.data!.tracks, (tr: TrackDto) => {
          ListItem() {
            Row() {
              Text(tr.title).layoutWeight(1).fontSize(15);
              Text(formatDuration((tr.duration ?? 0) * 1000))
                .fontSize(12).fontColor(this.theme.colors.textSecondary);
            }.padding(12);
          }
          .onClick(async () => {
            const tracks: Track[] = [];
            for (const x of this.data!.tracks) tracks.push(await resolveTrackToPlayer(x, false));
            const idx = this.data!.tracks.findIndex((x) => x.id === tr.id);
            await playerStore.playTrackList(tracks, idx >= 0 ? idx : 0);
            router.pushUrl({ url: 'pages/PlayerPage' });
          });
        }, (tr: TrackDto) => tr.id);
      }.layoutWeight(1);
    }.padding(16);
  }

  private async load(): Promise<void> {
    this.loading = true; this.error = '';
    try {
      this.data = await albumApi.getById(this.id) as AlbumDetail;
    } catch (e) {
      this.error = e instanceof Error ? e.message : String(e);
    } finally {
      this.loading = false;
    }
  }
}
```

```typescript
import { Logger } from 'audiodock_common';
import { rdbStore } from 'features_storage';
import { httpClient } from 'features_network';
import { authStore } from '../context/AuthStore';

export type SyncKind =
  | 'like_track' | 'unlike_track' | 'like_album' | 'unlike_album' | 'history';

interface SyncTask {
  kind: SyncKind;
  payload: Record<string, string | number | null>;
  attempts: number;
}

export type EventKind = 'trackLike' | 'albumLike' | 'history';
export type EventPayload =
  | { id: string; liked: boolean }
  | { trackId: string; progress: number; source: string | null };

type Listener = (payload: EventPayload) => void;

const MAX_ATTEMPTS = 5;
const DRAIN_INTERVAL_MS = 5000;
const EMPTY_TOKEN: string | null = null;

export class LikesHistoryService {
  private queue: SyncTask[] = [];
  private listeners = new Map<EventKind, Set<Listener>>();
  private timer: number | null = null;
  private draining: boolean = false;
  private currentTask: SyncTask | null = null;

  constructor() {
    this.start();
  }

  start(): void {
    if (this.timer !== null) return;
    this.timer = setInterval(() => {
      this.drain().catch((e: Error) => Logger.w('LikesHistory', `drain: ${e.message}`));
    }, DRAIN_INTERVAL_MS) as unknown as number;
  }

  stop(): void {
    if (this.timer !== null) {
      clearInterval(this.timer);
      this.timer = null;
    }
  }

  // --- Track likes -------------------------------------------------------
  async likeTrack(id: string): Promise<void> {
    await rdbStore.likeTrack(id);
    this.enqueue({ kind: 'like_track', payload: { trackId: id }, attempts: 0 });
    this.emit('trackLike', { id, liked: true });
  }

  async unlikeTrack(id: string): Promise<void> {
    await rdbStore.unlikeTrack(id);
    this.enqueue({ kind: 'unlike_track', payload: { trackId: id }, attempts: 0 });
    this.emit('trackLike', { id, liked: false });
  }

  async isTrackLiked(id: string): Promise<boolean> {
    return rdbStore.isTrackLiked(id);
  }

  async listLikedTracks() {
    return rdbStore.listLikedTracks();
  }

  // --- Album likes -------------------------------------------------------
  async likeAlbum(id: string): Promise<void> {
    await rdbStore.likeAlbum(id);
    this.enqueue({ kind: 'like_album', payload: { albumId: id }, attempts: 0 });
    this.emit('albumLike', { id, liked: true });
  }

  async unlikeAlbum(id: string): Promise<void> {
    await rdbStore.unlikeAlbum(id);
    this.enqueue({ kind: 'unlike_album', payload: { albumId: id }, attempts: 0 });
    this.emit('albumLike', { id, liked: false });
  }

  async isAlbumLiked(id: string): Promise<boolean> {
    return rdbStore.isAlbumLiked(id);
  }

  async listLikedAlbums() {
    return rdbStore.listLikedAlbums();
  }

  // --- History -----------------------------------------------------------
  async recordHistory(row: { track_id: string; progress: number; source: string | null }): Promise<void> {
    await rdbStore.upsertHistory({
      track_id: row.track_id,
      progress: row.progress,
      played_at: Date.now(),
      source: row.source,
    });
    this.enqueue({
      kind: 'history',
      payload: {
        trackId: row.track_id,
        progress: row.progress,
        playedAt: Date.now(),
        source: row.source ?? '',
      },
      attempts: 0,
    });
    this.emit('history', { trackId: row.track_id, progress: row.progress, source: row.source });
  }

  async listHistory(limit: number) {
    return rdbStore.listHistory(limit);
  }

  // --- Event bus ---------------------------------------------------------
  on(kind: EventKind, cb: Listener): () => void {
    if (!this.listeners.has(kind)) this.listeners.set(kind, new Set());
    this.listeners.get(kind)!.add(cb);
    return () => {
      const set = this.listeners.get(kind);
      if (set) set.delete(cb);
    };
  }

  // --- Sync queue --------------------------------------------------------
  enqueue(task: SyncTask): void {
    this.queue.push(task);
  }

  queueSize(): number {
    return this.queue.length + (this.currentTask ? 1 : 0);
  }

  async drain(): Promise<void> {
    if (this.draining) return;
    this.draining = true;
    try {
      const token = authStore.state_.token;
      if (!token) {
        if (this.queue.length > 0) {
          Logger.w('LikesHistory', `no auth token, dropping ${this.queue.length} queued tasks`);
          this.queue = [];
        }
        return;
      }
      while (this.queue.length > 0) {
        const task = this.queue[0];
        this.currentTask = task;
        try {
          await this.send(task);
          this.queue.shift();
        } catch (e) {
          task.attempts += 1;
          Logger.w('LikesHistory', `task ${task.kind} attempt ${task.attempts} failed: ${String(e)}`);
          if (task.attempts >= MAX_ATTEMPTS) {
            Logger.w('LikesHistory', `dropping ${task.kind} after ${MAX_ATTEMPTS} attempts`);
            this.queue.shift();
          } else {
            // leave at head, bail out so next tick retries
            break;
          }
        } finally {
          this.currentTask = null;
        }
      }
    } finally {
      this.draining = false;
    }
  }

  private async send(task: SyncTask): Promise<void> {
    switch (task.kind) {
      case 'like_track':   return httpClient.post('/likes/track', { liked: true,  ...task.payload });
      case 'unlike_track': return httpClient.post('/likes/track', { liked: false, ...task.payload });
      case 'like_album':   return httpClient.post('/likes/album', { liked: true,  ...task.payload });
      case 'unlike_album': return httpClient.post('/likes/album', { liked: false, ...task.payload });
      case 'history':      return httpClient.post('/track-history', task.payload);
    }
  }

  private emit(kind: EventKind, payload: EventPayload): void {
    const set = this.listeners.get(kind);
    if (!set) return;
    for (const cb of set) {
      try { cb(payload); } catch (e) {
        Logger.w('LikesHistory', `listener for ${kind} threw: ${String(e)}`);
      }
    }
  }
}

export const likesHistory = new LikesHistoryService();
```
